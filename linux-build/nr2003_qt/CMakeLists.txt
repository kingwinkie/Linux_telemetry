cmake_minimum_required(VERSION 3.16)
project(nr2003_qt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(nr2003_qt main.cpp mainwindow.cpp)
target_include_directories(nr2003_qt PRIVATE .. .)
target_link_libraries(nr2003_qt PRIVATE Qt6::Widgets)
